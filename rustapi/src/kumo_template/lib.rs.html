29/// Returns true when `source` contains none of the expansion markers of
30/// `dialect`, meaning the engine would render it with no substitution.
31fn source_is_static(dialect: TemplateDialect, source: &str) -> bool {
32    match dialect {
33        TemplateDialect::Static => true,
34        TemplateDialect::Handlebars => !source.contains("{{"),
35        TemplateDialect::Jinja => {
36            !(source.contains("{{") || source.contains("{%") || source.contains("{#"))
37        }
38    }
39}

41/// Returns the prefix of `source` that `dialect` renders for a static value:
42/// for Jinja, `source` with one trailing `\n` and then one trailing `\r`
43/// removed, matching the trim the minijinja lexer applies to every template it
44/// compiles. For the other dialects, `source` unchanged.
45fn trim_static_source(dialect: TemplateDialect, source: &str) -> &str {
46    if dialect == TemplateDialect::Jinja {
47        let s = source.strip_suffix('\n').unwrap_or(source);
48        s.strip_suffix('\r').unwrap_or(s)
49    } else {
50        source
51    }
52}

63impl<'env, 'source> Template<'env, 'source> {
64    /// Returns true when this template renders its source verbatim, ignoring
65    /// the context.
66    pub fn is_static(&self) -> bool {
67        matches!(self, Self::Static(_))
68    }
69
70    pub fn render<S: Serialize>(&self, ctx: S) -> anyhow::Result<String> {
71        match &self {
72            Self::Jinja(t) => Ok(t.render(ctx)?),
73            Self::Static(s) => Ok(s.to_string()),
74            Self::Handlebars { .. } => {
75                let mut output: Vec<u8> = vec![];
76                self.render_to_write(&ctx, &mut output)?;
77                Ok(String::from_utf8(output)?)
78            }
79        }
80    }
81
82    pub fn render_to_write<S: Serialize, W: std::io::Write>(
83        &self,
84        ctx: S,
85        mut w: W,
86    ) -> anyhow::Result<()> {
87        match &self {
88            Self::Jinja(t) => {
89                t.render_captured_to(ctx, w)?;
90                Ok(())
91            }
92            Self::Static(s) => {
93                w.write_all(s.as_bytes())?;
94                Ok(())
95            }
96            Self::Handlebars { engine, template } => {
97                let Engine::Handlebars { registry, globals } = &engine.engine else {
98                    anyhow::bail!("impossible Handlebars Template vs. TemplateEngine state")
99                };
100
101                let context = serde_json::to_value(ctx)?;
102                let context = merge_contexts(globals, context);
103                let context = handlebars::Context::wraps(context)?;
104
105                let output = render_handlebars(registry, template, &context)?;
106                w.write_all(output.as_bytes())?;
107                Ok(())
108            }
109        }
110    }
111
112    /// Renders this template against a context already converted to the native
113    /// form of the engine, returning the rendered output.
114    pub fn render_prepared(&self, ctx: &PreparedContext) -> anyhow::Result<String> {
115        match (self, ctx) {
116            (Self::Jinja(t), PreparedContext::Jinja(value)) => Ok(t.render(value)?),
117            (Self::Static(s), _) => Ok(s.to_string()),
118            (Self::Handlebars { engine, template }, PreparedContext::Handlebars(context)) => {
119                let Engine::Handlebars { registry, .. } = &engine.engine else {
120                    anyhow::bail!("impossible Handlebars Template vs. TemplateEngine state")
121                };
122                render_handlebars(registry, template, context)
123            }
124            _ => anyhow::bail!("prepared context dialect does not match template dialect"),
125        }
126    }
127}

129/// Renders `template` against `context` with the escaping rule the registry
130/// expects: substitutions are html-escaped only for a `.html` template name.
131fn render_handlebars(
132    registry: &Handlebars<'static>,
133    template: &HandlebarsTemplate,
134    context: &handlebars::Context,
135) -> anyhow::Result<String> {
136    let mut render_context = handlebars::RenderContext::new(None);
137    render_context.set_recursive_lookup(true);
138    let is_html = template
139        .name
140        .as_deref()
141        .map(|name| name.ends_with(".html"))
142        .unwrap_or(false);
143    render_context.set_disable_escape(!is_html);
144
145    Ok(template.renders(registry, context, &mut render_context)?)
146}

169    pub fn with_dialect(dialect: TemplateDialect) -> Self {
170        let engine = match dialect {
171            TemplateDialect::Jinja => {
172                let mut env = Environment::new();
173                env.set_unknown_method_callback(
174                    minijinja_contrib::pycompat::unknown_method_callback,
175                );
176                add_to_environment(&mut env);
177
178                env.add_filter(
179                    "normalize_smtp_response",
180                    mod_smtp_response_normalize::normalize,
181                );
182
183                Engine::Jinja { env }
184            }
185            TemplateDialect::Static => Engine::Static,
186            TemplateDialect::Handlebars => {
187                let mut registry = Handlebars::new();
188                registry.set_recursive_lookup(true);
189                Engine::Handlebars {
190                    registry,
191                    globals: HashMap::new(),
192                }
193            }
194        };
195        Self {
196            dialect,
197            engine,
198            literals: HashMap::new(),
199        }
200    }

202    /// Add a named template with the specified source. If name ends with
203    /// `.html` then automatic escaping of html entities will be performed on
204    /// substitutions.
205    ///
206    /// A source that is free of expansion markers for the active dialect is
207    /// stored as a literal static template, bypassing the compiler and
208    /// evaluator of the engine entirely.
209    pub fn add_template<N, S>(&mut self, name: N, source: S) -> anyhow::Result<()>
210    where
211        N: Into<String>,
212        S: Into<String>,
213    {
214        let name = name.into();
215        let mut source = source.into();
216
217        if source_is_static(self.dialect, &source) {
218            source.truncate(trim_static_source(self.dialect, &source).len());
219            self.literals.insert(name, source);
220            return Ok(());
221        }
222
223        match &mut self.engine {
224            Engine::Jinja { env } => {
225                let source: Cow<'_, str> = source.into();
226
227                Ok(env
228                    .add_template_owned(name, source.clone())
229                    .map_err(|err| {
230                        let mut reason = String::new();
231
232                        if let Some(detail) = err.detail() {
233                            write!(&mut reason, "{}: {}", err.kind(), detail).ok();
234                        } else {
235                            write!(&mut reason, "{}", err.kind()).ok();
236                        }
237
238                        if let Some((line_no, source_line)) = err.line().and_then(|line| {
239                            source
240                                .lines()
241                                .nth(line - 1) // err.line() is 1-based
242                                .map(|source_line| (line, source_line))
243                        }) {
244                            let truncated_line =
245                                &source_line[..source_line.ceil_char_boundary(1024)];
246
247                            if let Some(name) = err.name() {
248                                write!(
249                                    &mut reason,
250                                    " (in template '{name}' line {line_no}: '{truncated_line}')"
251                                )
252                                .ok();
253                            } else {
254                                write!(
255                                    &mut reason,
256                                    " (in template line {line_no}: '{truncated_line}')"
257                                )
258                                .ok();
259                            }
260                        }
261
262                        anyhow::anyhow!("{reason}")
263                    })?)
264            }
265            Engine::Static => anyhow::bail!("Static dialect cannot hold a dynamic template"),
266            Engine::Handlebars { registry, .. } => {
267                registry.register_template_string(&name, source)?;
268                Ok(())
269            }
270        }
271    }

273    /// Get a reference to a named template
274    pub fn get_template(&self, name: &str) -> anyhow::Result<Template<'_, '_>> {
275        if let Some(source) = self.literals.get(name) {
276            return Ok(Template::Static(source));
277        }
278        match &self.engine {
279            Engine::Jinja { env } => Ok(Template::Jinja(env.get_template(name)?)),
280            Engine::Static => anyhow::bail!("template {name} is not defined"),
281            Engine::Handlebars { registry, .. } => {
282                let template = registry
283                    .get_template(name)
284                    .ok_or_else(|| anyhow::anyhow!("template {name} is not defined"))?;
285
286                Ok(Template::Handlebars {
287                    engine: self,
288                    template,
289                })
290            }
291        }
292    }

294    /// Converts a context into the native form of the active dialect, merging in
295    /// the globals previously added with [`TemplateEngine::add_global`].
296    pub fn prepare_context<CTX>(&self, context: CTX) -> anyhow::Result<PreparedContext>
297    where
298        CTX: Serialize,
299    {
300        match &self.engine {
301            Engine::Jinja { .. } => Ok(PreparedContext::Jinja(JinjaValue::from_serialize(context))),
302            Engine::Static => Ok(PreparedContext::Static),
303            Engine::Handlebars { globals, .. } => {
304                let context = serde_json::to_value(context)?;
305                let context = merge_contexts(globals, context);
306                Ok(PreparedContext::Handlebars(handlebars::Context::wraps(
307                    context,
308                )?))
309            }
310        }
311    }

313    /// Define a global value that can be reference by all templates
314    pub fn add_global<N, V>(&mut self, name: N, value: V) -> anyhow::Result<()>
315    where
316        N: Into<String>,
317        V: Serialize,
318    {
319        match &mut self.engine {
320            Engine::Jinja { env } => env.add_global(name.into(), JinjaValue::from_serialize(value)),
321            Engine::Static => { /* NOP */ }
322            Engine::Handlebars { globals, .. } => {
323                globals.insert(name.into(), serde_json::to_value(value)?);
324            }
325        }
326
327        Ok(())
328    }
329
330    pub fn render<CTX>(&self, name: &str, source: &str, context: CTX) -> anyhow::Result<String>
331    where
332        CTX: serde::Serialize,
333    {
334        if source_is_static(self.dialect, source) {
335            return Ok(trim_static_source(self.dialect, source).to_string());
336        }
337        match &self.engine {
338            Engine::Jinja { env } => Ok(env.render_named_str(name, source, context)?),
339            Engine::Static => Ok(source.to_string()),
340            Engine::Handlebars { .. } => {
341                let template = HandlebarsTemplate::compile_with_name(source, name.to_string())?;
342                let template = Template::Handlebars {
343                    engine: self,
344                    template: &template,
345                };
346
347                template.render(context)
348            }
349        }
350    }
351}

353fn merge_contexts(
354    globals: &HashMap<String, serde_json::Value>,
355    over: serde_json::Value,
356) -> serde_json::Value {
357    match over {
358        serde_json::Value::Object(mut obj) => {
359            for (k, v) in globals {
360                if !obj.contains_key(k) {
361                    obj.insert(k.into(), v.clone());
362                }
363            }
364            serde_json::Value::Object(obj)
365        }
366        other => other,
367    }
368}
