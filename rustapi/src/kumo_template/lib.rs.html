46/// Converts a minijinja compile error into an error that names the template
47/// and quotes the offending line of `source`.
48fn describe_jinja_error(err: minijinja::Error, source: &str) -> anyhow::Error {
49    let mut reason = String::new();
50
51    if let Some(detail) = err.detail() {
52        write!(&mut reason, "{}: {}", err.kind(), detail).ok();
53    } else {
54        write!(&mut reason, "{}", err.kind()).ok();
55    }
56
57    if let Some((line_no, source_line)) = err.line().and_then(|line| {
58        source
59            .lines()
60            .nth(line - 1) // err.line() is 1-based
61            .map(|source_line| (line, source_line))
62    }) {
63        let truncated_line = &source_line[..source_line.ceil_char_boundary(1024)];
64
65        if let Some(name) = err.name() {
66            write!(
67                &mut reason,
68                " (in template '{name}' line {line_no}: '{truncated_line}')"
69            )
70            .ok();
71        } else {
72            write!(
73                &mut reason,
74                " (in template line {line_no}: '{truncated_line}')"
75            )
76            .ok();
77        }
78    }
79
80    anyhow::anyhow!("{reason}")
81}

83/// Returns true when `source` contains none of the expansion markers of
84/// `dialect`, meaning the engine would render it with no substitution.
85fn source_is_static(dialect: TemplateDialect, source: &str) -> bool {
86    match dialect {
87        TemplateDialect::Static => true,
88        TemplateDialect::Handlebars => !source.contains("{{"),
89        TemplateDialect::Jinja => {
90            !(source.contains("{{") || source.contains("{%") || source.contains("{#"))
91        }
92    }
93}

95/// Returns the prefix of `source` that `dialect` renders for a static value:
96/// for Jinja, `source` with one trailing `\n` and then one trailing `\r`
97/// removed, matching the trim the minijinja lexer applies to every template it
98/// compiles. For the other dialects, `source` unchanged.
99fn trim_static_source(dialect: TemplateDialect, source: &str) -> &str {
100    if dialect == TemplateDialect::Jinja {
101        let s = source.strip_suffix('\n').unwrap_or(source);
102        s.strip_suffix('\r').unwrap_or(s)
103    } else {
104        source
105    }
106}

108/// Renders `template` against `context` with the escaping rule the registry
109/// expects: substitutions are html-escaped only for a `.html` template name.
110fn render_handlebars(
111    registry: &Handlebars<'static>,
112    template: &HandlebarsTemplate,
113    context: &handlebars::Context,
114) -> anyhow::Result<String> {
115    let mut render_context = handlebars::RenderContext::new(None);
116    render_context.set_recursive_lookup(true);
117    let is_html = template
118        .name
119        .as_deref()
120        .map(|name| name.ends_with(".html"))
121        .unwrap_or(false);
122    render_context.set_disable_escape(!is_html);
123
124    Ok(template.renders(registry, context, &mut render_context)?)
125}

149/// A Jinja render context that converts its value on the first lookup, at
150/// most once, and also defines a global value named `alias` that returns
151/// that value.
152struct LazyContext<CTX> {
153    inner: Arc<CTX>,
154    alias: String,
155    converted: OnceLock<JinjaValue>,
156}
157
158impl<CTX: Serialize> LazyContext<CTX> {
159    fn value(&self) -> &JinjaValue {
160        self.converted
161            .get_or_init(|| JinjaValue::from_serialize(&*self.inner))
162    }
163}
164
165impl<CTX> std::fmt::Debug for LazyContext<CTX> {
166    fn fmt(&self, fmt: &mut std::fmt::Formatter) -> std::fmt::Result {
167        fmt.debug_struct("LazyContext")
168            .field("alias", &self.alias)
169            .field("converted", &self.converted.get().is_some())
170            .finish()
171    }
172}
173
174impl<CTX: Serialize + Send + Sync + 'static> Object for LazyContext<CTX> {
175    fn get_value(self: &Arc<Self>, key: &JinjaValue) -> Option<JinjaValue> {
176        let key = key.as_str()?;
177        let value = self.value();
178        if value.kind() == ValueKind::Invalid {
179            // Return the `Invalid` value to ensure that the error is
180            // not rendered as an empty value.
181            return Some(value.clone());
182        }
183        if key == self.alias {
184            // The alias is explicitly defined as returning the entire value,
185            // taking precedence over a hypothetical conflicting field with
186            // the same name.
187            return Some(value.clone());
188        }
189        // A missing name must yield None rather than undefined, otherwise the
190        // engine would stop short of its own globals such as `range`.
191        value
192            .get_attr(key)
193            .ok()
194            .filter(|value| !value.is_undefined())
195    }
196
197    fn enumerate(self: &Arc<Self>) -> Enumerator {
198        match self.value().try_iter() {
199            Ok(keys) => Enumerator::Values(
200                keys.filter(|key| key.as_str() != Some(self.alias.as_str()))
201                    .chain(std::iter::once(JinjaValue::from(self.alias.as_str())))
202                    .collect(),
203            ),
204            Err(_) => Enumerator::NonEnumerable,
205        }
206    }
207}

249impl Template {
250    /// Compiles `source` as a template of `dialect` named `name`. A name
251    /// ending in `.html` enables html entity escaping of substitutions.
252    /// If `source` is free of expansion markers, a Static Template is
253    /// returned.
254    fn compile(dialect: TemplateDialect, name: &str, source: &str) -> anyhow::Result<Self> {
255        let compiled = if source_is_static(dialect, source) {
256            Compiled::Static(trim_static_source(dialect, source).to_string())
257        } else {
258            match dialect {
259                TemplateDialect::Static => Compiled::Static(source.to_string()),
260                TemplateDialect::Jinja => {
261                    let mut env = new_jinja_environment();
262                    env.add_template_owned(name.to_string(), source.to_string())
263                        .map_err(|err| describe_jinja_error(err, source))?;
264                    Compiled::Jinja(OwnedJinja::try_new(env, |env| env.get_template(name))?)
265                }
266                TemplateDialect::Handlebars => Compiled::Handlebars(
267                    HandlebarsTemplate::compile_with_name(source, name.to_string())?,
268                ),
269            }
270        };
271
272        Ok(Self(Arc::new(TemplateInner {
273            dialect,
274            name: name.to_string(),
275            compiled,
276        })))
277    }
278
279    pub fn dialect(&self) -> TemplateDialect {
280        self.0.dialect
281    }
282
283    pub fn name(&self) -> &str {
284        &self.0.name
285    }
286
287    /// Returns true when this template renders its source verbatim
288    pub fn is_static(&self) -> bool {
289        matches!(self.0.compiled, Compiled::Static(_))
290    }
291
292    /// Renders this template against `ctx`, returning the rendered output.
293    pub fn render<S: Serialize>(&self, ctx: S) -> anyhow::Result<String> {
294        if let Compiled::Static(s) = &self.0.compiled {
295            return Ok(s.clone());
296        }
297        self.render_prepared(&PreparedContext::new(self.0.dialect, ctx)?)
298    }
299
300    /// Renders this template against a prepared context,
301    /// returning the rendered output.
302    pub fn render_prepared(&self, ctx: &PreparedContext) -> anyhow::Result<String> {
303        match (&self.0.compiled, ctx) {
304            (Compiled::Static(s), _) => Ok(s.clone()),
305            (Compiled::Jinja(cell), PreparedContext::Jinja(value)) => {
306                Ok(cell.borrow_dependent().render(value)?)
307            }
308            (Compiled::Handlebars(template), PreparedContext::Handlebars(context)) => {
309                render_handlebars(&HANDLEBARS_REGISTRY, template, context)
310            }
311            _ => anyhow::bail!("prepared context dialect does not match template dialect"),
312        }
313    }
314}

333/// Identifies a compiled template in [`COMPILED_TEMPLATES`] by a digest of
334/// its dialect, name and source.
335#[derive(Clone, Debug, Hash, PartialEq, Eq)]
336struct TemplateKey([u8; 32]);
337
338impl TemplateKey {
339    fn new(dialect: TemplateDialect, name: &str, source: &str) -> Self {
340        let mut hasher = blake3::Hasher::new();
341        hasher.update(&[dialect as u8]);
342        hasher.update(&(name.len() as u64).to_le_bytes());
343        hasher.update(name.as_bytes());
344        hasher.update(source.as_bytes());
345        Self(hasher.finalize().into())
346    }
347}

379    /// Add a named template with the specified source. If name ends with
380    /// `.html` then automatic escaping of html entities will be performed on
381    /// substitutions.
382    ///
383    /// A source that is free of expansion markers for the active dialect is
384    /// stored as a literal static template, bypassing the compiler and
385    /// evaluator of the engine entirely.
386    pub fn add_template<N, S>(&mut self, name: N, source: S) -> anyhow::Result<()>
387    where
388        N: Into<String>,
389        S: AsRef<str>,
390    {
391        let name = name.into();
392        let template = Template::compile(self.dialect, &name, source.as_ref())?;
393        self.templates.insert(name, template);
394        Ok(())
395    }
396
397    /// Returns the template registered under `name` with
398    /// [`TemplateEngine::add_template`].
399    pub fn get_template(&self, name: &str) -> anyhow::Result<Template> {
400        self.templates
401            .get(name)
402            .cloned()
403            .ok_or_else(|| anyhow::anyhow!("template {name} is not defined"))
404    }
405
406    /// Compiles `source` as a template named `name`, without registering it.
407    /// The compilation is cached.
408    pub async fn compile(&self, name: &str, source: &str) -> Result<Template, CompileError> {
409        let dialect = self.dialect;
410
411        // Static fragments are returned directly, as the copy is much cheaper
412        // than maintaining a cached entry.
413        if source_is_static(dialect, source) {
414            return Template::compile(dialect, name, source)
415                .map_err(|err| CompileError::Compile(Arc::new(err)));
416        }
417
418        let key = TemplateKey::new(dialect, name, source);
419        let lookup = COMPILED_TEMPLATES
420            .get_or_try_insert(&key, |_| COMPILED_TEMPLATE_TTL, async {
421                Template::compile(dialect, name, source).map_err(CompileFailure)
422            })
423            .await
424            .map_err(|err| {
425                if err.downcast_ref::<CompileFailure>().is_some() {
426                    CompileError::Compile(err)
427                } else {
428                    CompileError::Cache(err)
429                }
430            })?;
431        Ok(lookup.item)
432    }

434    /// Converts `context` into the native form for this engine's dialect,
435    /// producing a [`PreparedContext`] that can render more than one template
436    /// without being converted again.
437    pub fn prepare_context<CTX>(&self, context: CTX) -> anyhow::Result<PreparedContext>
438    where
439        CTX: Serialize,
440    {
441        PreparedContext::new(self.dialect, context)
442    }
443
444    /// Converts `context` as [`TemplateEngine::prepare_context`] does, and in
445    /// addition binds the whole converted value to a global variable
446    /// named `alias`.
447    pub fn prepare_context_with_alias<CTX>(
448        &self,
449        context: Arc<CTX>,
450        alias: &str,
451    ) -> anyhow::Result<PreparedContext>
452    where
453        CTX: Serialize + Send + Sync + 'static,
454    {
455        Ok(match self.dialect {
456            TemplateDialect::Jinja => {
457                PreparedContext::Jinja(JinjaValue::from_object(LazyContext {
458                    inner: context,
459                    alias: alias.to_string(),
460                    converted: OnceLock::new(),
461                }))
462            }
463            TemplateDialect::Static => PreparedContext::Static,
464            TemplateDialect::Handlebars => {
465                let mut root = serde_json::to_value(&*context)?;
466                let whole = root.clone();
467                if let Some(fields) = root.as_object_mut() {
468                    fields.insert(alias.to_string(), whole);
469                }
470                PreparedContext::Handlebars(handlebars::Context::from(root))
471            }
472        })
473    }

475    /// Compiles `source` and renders it against `context` in one step.
476    pub fn render<CTX>(&self, name: &str, source: &str, context: CTX) -> anyhow::Result<String>
477    where
478        CTX: Serialize,
479    {
480        Template::compile(self.dialect, name, source)?.render(context)
481    }

891    /// Converts a Jinja aliased context value at most once, on the first name
892    /// lookup of any kind, however many fields and renders read it.
893    /// Names not defined in the value must still be resolved against
894    /// engine-provided globals.
895    #[test]
896    fn aliased_context_serializes_at_most_once() {
897        let mut engine = TemplateEngine::with_dialect(TemplateDialect::Jinja);
898        engine
899            .add_template("fields", "{{ type }} {{ size }} {{ log_record.size }}")
900            .unwrap();
901        engine
902            .add_template("globals", "{{ range(3) | length }}")
903            .unwrap();
904
905        let record = counting_record();
906        let prepared = engine
907            .prepare_context_with_alias(record.clone(), "log_record")
908            .unwrap();
909        let reads = || {
910            record
911                .serializations
912                .load(std::sync::atomic::Ordering::Relaxed)
913        };
914
915        k9::assert_equal!(reads(), 0);
916        let globals = engine.get_template("globals").unwrap();
917        k9::assert_equal!(globals.render_prepared(&prepared).unwrap(), "3".to_string());
918        k9::assert_equal!(reads(), 1);
919
920        let fields = engine.get_template("fields").unwrap();
921        for _ in 0..3 {
922            k9::assert_equal!(
923                fields.render_prepared(&prepared).unwrap(),
924                "Delivery 1047 1047".to_string()
925            );
926        }
927        k9::assert_equal!(reads(), 1);
