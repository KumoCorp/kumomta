1//! Coordinates write pauses and automatic retries after spool errors.
2
3use parking_lot::Mutex;
4use std::sync::atomic::{AtomicBool, Ordering};
5use std::time::{Duration, Instant};
6
7/// Timing and toggle that govern when the gate latches and when it may
8/// reopen on its own.
9#[derive(Clone, Copy)]
10pub(super) struct Policy {
11    /// Delay from the first error of an incident until the gate latches.
12    pub latch_duration: Duration,
13    /// Quiet interval required, once latched, before an automatic reopen.
14    pub unlatch_duration: Duration,
15    /// Whether the gate may reopen on its own. False requires a restart.
16    pub allow_unlatch: bool,
17}
18
19impl Policy {
20    /// Rejects a configuration that could never keep the gate open: with
21    /// unlatching enabled, a zero unlatch interval would clear and re-latch
22    /// on the same monitor tick.
23    pub fn validate(&self) -> anyhow::Result<()> {
24        anyhow::ensure!(
25            !self.allow_unlatch || !self.unlatch_duration.is_zero(),
26            "error_unlatch_duration must be greater than zero when allow_error_unlatch is true"
27        );
28        Ok(())
29    }
30}
31
32/// Serializes error observations and gate transitions while allowing
33/// lock-free checks.
34pub(super) struct Health {
35    state: Mutex<State>,
36    active: AtomicBool,
37    policy: Policy,
38    path: String,
39}
40
41impl Health {
42    pub fn new(background_errors: u64, policy: Policy, path: String) -> Self {
43        Self {
44            state: Mutex::new(State::new(background_errors)),
45            active: AtomicBool::new(false),
46            policy,
47            path,
48        }
49    }
50
51    pub fn is_active(&self) -> bool {
52        self.active.load(Ordering::Relaxed)
53    }
54
55    /// Returns the delay from the first observed error in an incident before
56    /// the gate latches.
57    pub fn latch_duration(&self) -> Duration {
58        self.policy.latch_duration
59    }
60
61    /// Records an error and returns whether it starts an incident or
62    /// latches the gate.
63    pub fn record_foreground_error(&self, fatal: bool) -> bool {
64        self.update(|state| state.foreground_error(Instant::now(), fatal))
65    }
66
67    /// Feeds the latest background-error count from the periodic monitor into
68    /// the state machine, driving the latch and reopen timers. Called on every
69    /// monitor tick with a fresh sample, not just when the count changed.
70    /// Growth over the previous sample is a new error.
71    pub fn sample_background_errors(&self, background_errors: u64) {
72        let mut growth = None;
73        self.update(|state| {
74            if background_errors > state.background_errors {
75                growth = Some((state.background_errors, background_errors));
76            }
77            state.tick(Instant::now(), background_errors, self.policy)
78        });
79        // Logged after `update` releases the state lock: `tracing` dispatch
80        // is synchronous and can block on a slow sink, and holding the lock
81        // across that would serialize every concurrent error reporter
82        // behind it.
83        if let Some((prev, new)) = growth {
84            tracing::error!(
85                "rocksdb at {}: background error count increased from {prev} to {new}; \
86                 check the LOG file in that directory for details",
87                self.path,
88            );
89        }
90    }
91
92    /// Runs one state mutation under the lock. It mirrors the result into the
93    /// lock-free gate and logs it. Returns true when the caller should log the
94    /// underlying cause: the mutation either started a fresh incident or
95    /// latched the gate.
96    fn update(&self, change: impl FnOnce(&mut State) -> Option<Transition>) -> bool {
97        let (started_incident, latched, log) = {
98            let mut state = self.state.lock();
99            let was_healthy = state.unhealthy_since.is_none();
100            let transition = change(&mut state);
101            let started_incident = was_healthy && state.unhealthy_since.is_some();
102            let latched = matches!(transition, Some(Transition::Latched { .. }));
103            // Store the mirrored flag and snapshot the log fields while
104            // still holding the lock. Otherwise a clear could overwrite the
105            // immediate latch from a concurrent fatal error, or the snapshot
106            // could observe a later state than the transition it describes.
107            let log = self.publish(&state, transition);
108            (started_incident, latched, log)
109        };
110        // Emit the log after releasing the lock: `tracing` dispatch is
111        // synchronous and can block on a slow sink, and holding the lock
112        // across that would serialize every concurrent error reporter
113        // behind it.
114        if let Some(log) = log {
115            self.log_transition(log);
116        }
117        started_incident || latched
118    }
119
120    /// Mirrors the latched flag into the lock-free `active` gate that the
121    /// hot write path reads. Returns the fields of a one-shot transition log
122    /// for a latch or clear, or `None` if `transition` is `None`.
123    fn publish(&self, state: &State, transition: Option<Transition>) -> Option<LogTransition> {
124        self.active
125            .store(state.latched_at.is_some(), Ordering::Relaxed);
126        match transition {
127            Some(Transition::Latched { fatal }) => Some(LogTransition::Latched {
128                fatal,
129                foreground_errors: state
130                    .foreground_errors
131                    .saturating_sub(state.foreground_baseline),
132                background_errors: state
133                    .background_errors
134                    .saturating_sub(state.background_baseline),
135                latch_delay: if fatal {
136                    Duration::ZERO
137                } else {
138                    self.policy.latch_duration
139                },
140            }),
141            Some(Transition::Cleared) => Some(LogTransition::Cleared {
142                unlatch_duration: self.policy.unlatch_duration,
143            }),
144            None => None,
145        }
146    }
147
148    /// Emits the one-shot transition log built by `publish`.
149    fn log_transition(&self, log: LogTransition) {
150        match log {
151            LogTransition::Latched {
152                fatal,
153                foreground_errors,
154                background_errors,
155                latch_delay,
156            } => tracing::error!(
157                "rocksdb at {}: load-shedding gate latched; fatal={fatal}, \
158                 foreground errors={foreground_errors}, \
159                 background errors={background_errors}, latch delay={latch_delay:?}. \
160                 Ingress paths will reject traffic. Inspect the LOG file \
161                 for the underlying cause.",
162                self.path,
163            ),
164            LogTransition::Cleared { unlatch_duration } => tracing::info!(
165                "rocksdb at {}: load-shedding gate cleared after {:?} \
166                 latched and without newly observed errors; ingress may retry \
167                 the database. Database recovery has not been verified.",
168                self.path,
169                unlatch_duration,
170            ),
171        }
172    }
173}

175/// The fields of a one-shot transition log, snapshotted from `State` under
176/// the lock.
177enum LogTransition {
178    Latched {
179        fatal: bool,
180        foreground_errors: u64,
181        background_errors: u64,
182        latch_delay: Duration,
183    },
184    Cleared {
185        unlatch_duration: Duration,
186    },
187}
188
189/// A change in the open/closed state of the gate that the caller should log
190/// once.
191#[derive(Debug, PartialEq, Eq)]
192enum Transition {
193    /// The gate closed to writes. `fatal` distinguishes the causes: a
194    /// Corruption or IOError that latched immediately, versus a milder signal
195    /// that latched only after `latch_duration` elapsed. It does not mean the
196    /// gate is permanent. A fatal latch can still reopen automatically.
197    Latched { fatal: bool },
198    /// The gate reopened to writes after the quiet interval.
199    Cleared,
200}
201
202/// Per-incident error tracking. Separates the errors of the current incident
203/// from counts retained from earlier incidents, and holds the timing that the
204/// latch and reopen decisions are made from.
205struct State {
206    /// Background-error count captured when the current incident's baseline
207    /// was set (construction or the last clear). Only growth above this
208    /// counts as an error in this incident.
209    background_baseline: u64,
210    /// Foreground-error count captured at the same point. Only growth above
211    /// it counts as a foreground error in this incident.
212    foreground_baseline: u64,
213    /// Highest background-error count seen so far. A failed property read
214    /// samples as zero. Keeping the maximum prevents that dip from looking like
215    /// the count moved backwards.
216    background_errors: u64,
217    /// Running count of foreground errors reported to this tracker.
218    foreground_errors: u64,
219    /// When the first error of the current incident was seen. None while
220    /// healthy. Drives the latch delay and is cleared on reopen.
221    unhealthy_since: Option<Instant>,
222    /// When the most recent error in the current incident was seen. The
223    /// reopen interval is measured from this (or the latch time, if later).
224    last_error_at: Option<Instant>,
225    /// When the gate latched. None while it is open.
226    latched_at: Option<Instant>,
227}
228
229impl State {
230    fn new(background_errors: u64) -> Self {
231        Self {
232            background_baseline: background_errors,
233            foreground_baseline: 0,
234            background_errors,
235            foreground_errors: 0,
236            unhealthy_since: None,
237            last_error_at: None,
238            latched_at: None,
239        }
240    }
241
242    /// Records that a fresh error occurred now: opens the incident if this
243    /// is its first error, and advances the last-error timestamp that the
244    /// reopen interval is measured from.
245    fn record_error(&mut self, now: Instant) {
246        self.unhealthy_since.get_or_insert(now);
247        self.last_error_at = Some(now);
248    }
249
250    /// Reports a foreground (load/store/remove/enumerate) error. A fatal class
251    /// latches the gate immediately. Anything else only starts (or extends) the
252    /// incident and leaves latching to the delayed path in `tick`. Returns the
253    /// latch transition when this call caused it.
254    fn foreground_error(&mut self, now: Instant, fatal: bool) -> Option<Transition> {
255        self.foreground_errors += 1;
256        self.record_error(now);
257        if fatal && self.latched_at.is_none() {
258            self.latched_at = Some(now);
259            return Some(Transition::Latched { fatal: true });
260        }
261        None
262    }
263
264    /// Advances the state machine on a periodic monitor sample: folds in the
265    /// latest background-error count, then applies whichever timer is due --
266    /// reopen if latched and quiet long enough, otherwise latch if the
267    /// incident has run past `latch_duration`. Returns the resulting
268    /// transition, if any.
269    fn tick(&mut self, now: Instant, background_errors: u64, policy: Policy) -> Option<Transition> {
270        if background_errors > self.background_errors {
271            self.record_error(now);
272        }
273        self.background_errors = self.background_errors.max(background_errors);
274
275        if let Some(latched_at) = self.latched_at {
276            let quiet_since = self.last_error_at.unwrap_or(latched_at).max(latched_at);
277            if policy.allow_unlatch && now.duration_since(quiet_since) >= policy.unlatch_duration {
278                self.latched_at = None;
279                self.unhealthy_since = None;
280                self.last_error_at = None;
281                self.background_baseline = self.background_errors;
282                self.foreground_baseline = self.foreground_errors;
283                return Some(Transition::Cleared);
284            }
285        } else if let Some(since) = self.unhealthy_since {
286            // Latch even for an isolated error, since a stuck database can stop
287            // reporting errors entirely. RocksDB can pause background
288            // scheduling after a paranoid-check failure, dropping
289            // compaction-pending and is-write-stopped to zero. Use error
290            // history to cover that case.
291            if now.duration_since(since) >= policy.latch_duration {
292                self.latched_at = Some(now);
293                return Some(Transition::Latched { fatal: false });
294            }
295        }
296        None
297    }
298}

328    /// `State::new` takes a starting background-error count as its baseline,
329    /// and only growth above it opens an incident. This checks the non-growth
330    /// cases against a nonzero baseline: re-observing the baseline, and a dip
331    /// to zero from a failed property read, neither of which opens an incident.
332    /// `RocksSpool::new` always baselines at zero in practice, but `State`
333    /// itself accepts any starting count.
334    #[test]
335    fn nonzero_baseline_does_not_start_an_incident_on_its_own() {
336        let now = Instant::now();
337        let mut state = State::new(7);
338        for (seconds, bg) in [(0, 7), (60, 0), (120, 7)] {
339            assert_equal!(
340                state.tick(now + Duration::from_secs(seconds), bg, policy()),
341                None
342            );
343            assert_equal!(state.unhealthy_since, None);
344            assert_equal!(state.latched_at, None);
345        }
346        assert_equal!(state.background_errors, 7);
347    }

349    #[test]
350    fn zero_latch_delay() {
351        let mut policy = policy();
352        policy.latch_duration = Duration::ZERO;
353        let now = Instant::now();

355        assert_equal!(
356            state.tick(now, 1, policy),
357            Some(Transition::Latched { fatal: false })
358        );

362    /// One error of one signal (foreground or background, with the other
363    /// silent) latches after `latch_duration` and reopens after the quiet
364    /// interval. Running the cycle three times, an hour apart, checks that each
365    /// incident is measured on its own and gets a fresh delay rather than
366    /// inheriting a stale timestamp from the previous one.
367    #[test]
368    fn isolated_errors_and_repeated_incidents() {
369        for foreground in [false, true] {
370            let mut state = State::new(7);
371            let start = Instant::now();
372            let mut bg = 7;
373            assert_equal!(state.tick(start, bg, policy()), None);
374            for incident in 0..3 {
375                let now = start + Duration::from_secs(incident * 3600);
376                if foreground {
377                    assert_equal!(state.foreground_error(now, false), None);
378                } else {
379                    bg += 1;
380                }
381                assert_equal!(state.tick(now, bg, policy()), None);
382                assert_equal!(
383                    state.tick(now + Duration::from_secs(14), bg, policy()),
384                    None
385                );
386                assert_equal!(
387                    state.tick(now + Duration::from_secs(15), bg, policy()),
388                    Some(Transition::Latched { fatal: false })
389                );
390                assert_equal!(
391                    state.tick(now + Duration::from_secs(44), bg, policy()),
392                    None
393                );
394                assert_equal!(
395                    state.tick(now + Duration::from_secs(45), bg, policy()),
396                    Some(Transition::Cleared)
397                );
398                assert_equal!(
399                    state.tick(now + Duration::from_secs(3599), bg, policy()),
400                    None
401                );
402                assert_equal!(state.latched_at, None);
403            }
404            assert_equal!(state.foreground_errors, if foreground { 3 } else { 0 });
405        }
406    }

492    #[test]
493    fn fatal_error_and_clear_are_serialized() {
494        for _ in 0..100 {
495            let health = Arc::new(Health::new(0, policy(), "test".to_string()));
496            let start = Instant::now();
497            let retry_at = start + Duration::from_secs(60);
498            health.update(|state| state.foreground_error(start, true));
499            let barrier = Arc::new(Barrier::new(2));

503                    health.update(|state| state.foreground_error(retry_at, true));

507            });
508            assert_equal!(health.is_active(), true);
